const { createApp, ref } = Vue;

const app = createApp({
    setup() {
        const isLoggedIn = ref(false);
        const loading = ref(false);

        // 新增：登录表单
        const username = ref('');
        const password = ref('');

        const scoreSolid = ref(0);
        const scoreStripe = ref(0);
        const foul = ref(false);
        const isPaused = ref(false);

        let pollingTimer = null;
        let gameOverShown = false;

        const handleLogin = () => {
            if (loading.value || isLoggedIn.value) return;

            if (!username.value.trim() || !password.value.trim()) {
                ElementPlus.ElMessage.warning('请输入用户名和密码');
                return;
            }

            loading.value = true;

            axios.post('/api/login', {
                username: username.value.trim(),
                password: password.value.trim()
            })
            .then(res => {
                if (res.data.ok) {
                    // 给淡出动画一点时间
                    setTimeout(() => {
                        isLoggedIn.value = true;
                        loading.value = false;
                        startPolling();
                    }, 220);
                } else {
                    loading.value = false;
                    ElementPlus.ElMessage.error(res.data.msg || '登录失败');
                }
            })
            .catch(err => {
                loading.value = false;
                const msg = err?.response?.data?.msg || '登录失败';
                ElementPlus.ElMessage.error(msg);
                console.error('登录失败:', err);
            });
        };

        const startPolling = () => {
            if (pollingTimer) return;

            fetchStatus(); // 先拉一次
            pollingTimer = setInterval(fetchStatus, 500);
        };

        const fetchStatus = () => {
            axios.get('/api/status')
                .then(res => {
                    scoreSolid.value = res.data.solid;
                    scoreStripe.value = res.data.stripe;
                    foul.value = res.data.foul;

                    if (res.data.game_over) {
                        if (!gameOverShown) {
                            ElementPlus.ElMessage.warning('黑八入袋，比赛重置！');
                            gameOverShown = true;
                        }
                    } else {
                        gameOverShown = false;
                    }
                })
                .catch(err => {
                    console.error('获取状态失败:', err);
                });
        };

        const togglePause = () => {
            if (isPaused.value) {
                axios.post('/api/control/resume')
                    .then(() => { isPaused.value = false; })
                    .catch(err => {
                        ElementPlus.ElMessage.error('恢复播放失败');
                        console.error(err);
                    });
            } else {
                axios.post('/api/control/pause')
                    .then(() => { isPaused.value = true; })
                    .catch(err => {
                        ElementPlus.ElMessage.error('暂停视频失败');
                        console.error(err);
                    });
            }
        };

        const restartVideo = () => {
            if (confirm('确定要重新开始比赛并重播视频吗？')) {
                axios.post('/api/control/restart')
                    .then(() => {
                        isPaused.value = false;
                        foul.value = false;
                        gameOverShown = false;
                    })
                    .catch(err => {
                        ElementPlus.ElMessage.error('重播失败');
                        console.error(err);
                    });
            }
        };

        const adjust = (team, delta) => {
            axios.post('/api/adjust', { team, delta })
                .then(() => {
                    // 保持原来即时反馈体验
                    if (team === 'solid') scoreSolid.value += delta;
                    else scoreStripe.value += delta;
                })
                .catch(err => {
                    ElementPlus.ElMessage.error('调整比分失败');
                    console.error(err);
                });
        };

        const clearFoul = () => {
            axios.get('/api/clear_foul')
                .then(() => { foul.value = false; })
                .catch(err => {
                    ElementPlus.ElMessage.error('清除犯规失败');
                    console.error(err);
                });
        };

        const resetGame = () => {
            if (confirm('仅清空比分？')) {
                axios.get('/api/reset')
                    .then(() => {
                        scoreSolid.value = 0;
                        scoreStripe.value = 0;
                        foul.value = false;
                        gameOverShown = false;
                    })
                    .catch(err => {
                        ElementPlus.ElMessage.error('重置比赛失败');
                        console.error(err);
                    });
            }
        };

        return {
            isLoggedIn, loading,
            username, password,
            handleLogin,
            scoreSolid, scoreStripe, foul, isPaused,
            togglePause, restartVideo, adjust, clearFoul, resetGame
        };
    }
});

app.config.compilerOptions.delimiters = ['[[', ']]'];
app.use(ElementPlus);
app.mount('#app');

MODEL_PATH = os.path.join(base_path, "models", "best.pt")

LOGIN_USER = "admin"
LOGIN_PASS = "123456"

video_state = {
    "is_paused": False,
    "need_restart": False
}

state_lock = threading.Lock()
frame_lock = threading.Lock()

latest_jpeg = None
worker_started = False

print(f"🚀 Loading Model: {MODEL_PATH}")
model = YOLO(MODEL_PATH)
print("模型类别:", model.names)  # 方便你确认黑八类别ID
scorer = BilliardScoring()


def video_worker():
    """后台唯一视频线程：只开一次视频和YOLO追踪"""
    global latest_jpeg

    cap = cv2.VideoCapture(VIDEO_SOURCE)
    if not cap.isOpened():
        print(f"❌ 无法打开视频: {VIDEO_SOURCE}")
        return

    fps = cap.get(cv2.CAP_PROP_FPS)
    if fps <= 0:
        fps = 30.0
    frame_delay = 1.0 / fps

    print(f"✅ 视频线程启动，FPS={fps:.2f}")

    while True:
        start_time = time.time()

        with state_lock:
            need_restart = video_state["need_restart"]
            is_paused = video_state["is_paused"]

            if need_restart:
                cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
                scorer.reset_game()
                video_state["need_restart"] = False
                video_state["is_paused"] = False
                is_paused = False

        if is_paused:
            time.sleep(0.05)
            continue

        success, frame = cap.read()
        if not success:
            cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
            time.sleep(0.01)
            continue

        try:
            results = model.track(
                frame,
                persist=True,
                conf=0.15,
                iou=0.4,
                tracker="bytetrack.yaml",
                verbose=False
            )

            scorer.update(results[0])
            annotated_frame = results[0].plot()

            # 画袋口监控圈
            for px, py in scorer.pockets:
                cv2.circle(
                    annotated_frame,
                    (int(px), int(py)),
                    scorer.pocket_radius,
                    (0, 255, 0),
                    2
                )

        except Exception as e:
            print(f"⚠️ YOLO处理异常: {e}")
            annotated_frame = frame

        ret, buffer = cv2.imencode('.jpg', annotated_frame)
        if ret:
            with frame_lock:
                latest_jpeg = buffer.tobytes()

        # 控制播放速度
        processing_time = time.time() - start_time
        sleep_time = frame_delay - processing_time
        if sleep_time > 0:
            time.sleep(sleep_time)


def ensure_worker_started():
    """确保后台线程只启动一次"""
    global worker_started
    with state_lock:
        if not worker_started:
            t = threading.Thread(target=video_worker, daemon=True)
            t.start()
            worker_started = True


def generate_stream():
    """前端只读取最新帧，不重复开YOLO/视频"""
    ensure_worker_started()

    while True:
        with frame_lock:
            frame_data = latest_jpeg

        if frame_data is None:
            time.sleep(0.03)
            continue

        yield (b'--frame\r\n'
               b'Content-Type: image/jpeg\r\n\r\n' + frame_data + b'\r\n')

        time.sleep(0.01)
